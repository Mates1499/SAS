\section{Klasifikace signálů ve spojitém a diskrétním čase}
% ============================================================

\QA{Co je to signál?}{%
  \textbf{Signál} je reálná nebo komplexní funkce nezávislé proměnné – nejčastěji čas $t$ nebo index vzorku $k$:
  \[
    s: \mathbb{R} \to \mathbb{R} \quad \text{(nebo } \mathbb{C}\text{)}
    \qquad
    s: \mathbb{Z} \to \mathbb{R} \quad \text{(nebo } \mathbb{C}\text{)}
  \]
  \begin{itemize}[itemsep=3pt]
    \item Signál popisuje veličinu, která se mění v čase (napětí, zvuk, elektromagnetická vlna, teplota).
    \item U obrazů jsou nezávislou proměnnou souřadnice $x, y$ – jde o dvoudimenzionální signál $s(x,y)$.
    \item Signál je matematickým modelem fyzikálního děje – nosič informace.
  \end{itemize}
  \begin{Poznamka}[Příklady]
    Jednorozměrné: zvuk $s(t)$, napětí $u(t)$, radiový signál. Dvourozměrné: černobílý obraz $s(x,y)$. Vícerozměrné: video, MRI.
  \end{Poznamka}
}

\QA{Jak vypadá signál ve spojitém a diskrétním čase?}{%
  \begin{Info}[Spojitý čas]
    Nezávislá proměnná $t \in \mathbb{R}$ – signál $s(t)$ je definován pro \textbf{každý} reálný časový okamžik. Má obvykle hladký průběh. Příklady: analogový zvuk, napětí na kondenzátoru.
  \end{Info}

  \begin{Info}[Diskrétní čas]
    Nezávislá proměnná $k \in \mathbb{Z}$ – signál $s[k]$ je definován jen v \textbf{celočíselných okamžicích}. Vzniká vzorkováním spojitého signálu s periodou $T_{sa}$:
    \[
      s[k] = s(kT_{sa}), \quad T_{sa} = \frac{1}{f_{sa}}
    \]
  \end{Info}

  Kombinace spojitého/diskrétního času a spojitých/diskrétních hodnot dává 4 typy signálů. \textbf{Vzorkování} převádí spojitý čas na diskrétní; \textbf{kvantizace} převádí spojité hodnoty na diskrétní (vzniká digitální signál).
}

\QA{Jak vypadají a jaké základní vlastnosti mají jednotkový skok, obdélníkový impuls, jednotkový impuls, Diracův impuls a vzorkovací funkce?}{%

  \textbf{1) Jednotkový skok} $1(t)$, resp.\ $u[k]$:
  \begin{Vzorec}[]
  \[
    1(t) = \begin{cases} 0 & t < 0 \\ \tfrac{1}{2} & t = 0 \\ 1 & t > 0 \end{cases}
    \qquad
    u[k] = \begin{cases} 0 & k < 0 \\ 1 & k \geq 0 \end{cases}
  \]
  \end{Vzorec}
  Vlastnosti: $\Av[1(t)] = \tfrac{1}{2}$, výkon $P = \tfrac{1}{2}$, ACF $R(\tau) = \tfrac{1}{2}$ (konstantní), výkonový signál.

  \vspace{0.5em}
  \textbf{2) Obdélníkový impuls} $\rect(t/\tau)$:
  \[
    \rect\!\left(\tfrac{t}{\tau}\right) = 1\!\left(t+\tfrac{\tau}{2}\right) - 1\!\left(t-\tfrac{\tau}{2}\right)
    \quad \Rightarrow \quad E = \tau
  \]
  Energetický signál. Složen ze dvou posunutých jednotkových skoků. FT: $\Sa(\omega\tau/2)$ – sinc funkce.

  \vspace{0.5em}
  \textbf{3) Jednotkový impuls} $\delta[k]$ (Kroneckerovo delta – diskrétní čas):
  \begin{Vzorec}[]
  \[
    \delta[k] = \begin{cases} 1 & k = 0 \\ 0 & k \neq 0 \end{cases}
  \]
  \end{Vzorec}
  Jde o Diracův impuls v diskrétním čase. Vzorkovací vlastnost: $\sum_k s[k]\,\delta[k-k_0] = s[k_0]$.

  \vspace{0.5em}
  \textbf{4) Diracův impuls} $\delta(t)$ (spojitý čas):
  \begin{Vzorec}[]
  \[
    \delta(t) = \frac{\dd}{\dd t}1(t),
    \qquad
    \int_{-\infty}^{+\infty}\!\delta(t)\,\dd t = 1 \quad \text{(jednotková plocha)}
  \]
  Vzorkovací (sifting) vlastnost: $\displaystyle\int_{-\infty}^{+\infty} s(t)\,\delta(t-t_0)\,\dd t = s(t_0)$
  \end{Vzorec}
  FT: $\FT\{\delta(t)\} = 1$ (konstantní spektrum – všechny frekvence jsou přítomny stejně).

  \vspace{0.5em}
  \textbf{5) Vzorkovací funkce} (sinc):
  \begin{Vzorec}[]
  \[
    \Sa(t) = \frac{\sin t}{t},\quad \Sa(0)=1
    \qquad
    \sinc(t) = \frac{\sin(\pi t)}{\pi t} \quad \text{(normalizovaná)}
  \]
  \end{Vzorec}
  Energie: $E_{\Sa}=\pi$. Nuly v $t=n\pi$, $n\in\mathbb{Z}\setminus\{0\}$. FT: obdélník ve spektru (duálnost s rect).

  \begin{Poznamka}[Duálnost FT]
    $\rect \xleftrightarrow{\FT} \Sa$ a $\delta \xleftrightarrow{\FT} 1$ – duální páry; co je obdélník v čase, je sinc ve spektru, a naopak.
  \end{Poznamka}
}

\section{Spektrální reprezentace, korelace, Fourierovy transformace}
% ============================================================

\QA{Co popisují a jaký je mechanismus výpočtu vzájemné korelační funkce a autokorelační funkce?}{%
  \textbf{Vzájemná korelační funkce} $R_{12}(\tau)$ popisuje \textbf{míru podobnosti} dvou různých signálů $s_1$ a $s_2$ při vzájemném posuvu $\tau$:
  \begin{Vzorec}[Spojitý čas]
  \[
    R_{12}(\tau) = \int_{-\infty}^{+\infty} s_1(t)\,s_2^*(t-\tau)\,\dd t
  \]
  \end{Vzorec}
  \begin{Vzorec}[Diskrétní čas]
  \[
    R_{12}[m] = \sum_{k=-\infty}^{\infty} s_1[k]\,s_2^*[k-m]
  \]
  \end{Vzorec}
  \textbf{Mechanismus}: signál $s_2$ postupně posouváme o $\tau$, pronásobujeme ho se $s_1$ a integrujeme. Hledáme posun $\tau$, při němž je vzájemná energie největší – signály si jsou nejpodobnější. Pro reálné signály: $R_{12}(\tau) = R_{21}(-\tau)$.

  \vspace{0.5em}
  \textbf{Autokorelační funkce} $R(\tau)$ popisuje \textbf{závislost signálu na sobě samém} s posunem $\tau$:
  \begin{Vzorec}[Energetický signál]
  \[
    R(\tau) = \int_{-\infty}^{+\infty} s(t)\,s^*(t-\tau)\,\dd t, \qquad R(0) = E
  \]
  \end{Vzorec}
  \begin{Vzorec}[Výkonový signál]
  \[
    R(\tau) = \Av\!\left[s(t+\tau)\,s^*(t)\right], \qquad R(0) = P
  \]
  \end{Vzorec}
  Klíčové vlastnosti ACF:
  \begin{itemize}[itemsep=2pt]
    \item \textbf{Sudá funkce}: $R(\tau) = R(-\tau)$ (pro reálné signály).
    \item \textbf{Globální maximum v $\tau=0$}: $|R(\tau)| \leq R(0)$.
    \item $R(0) = E$ nebo $R(0) = P$.
  \end{itemize}
  \begin{Poznamka}[Praktické využití]
    GPS – korelace přijatého signálu s referenčními kódy pro zjištění zpoždění. Radar – detekce odraženého pulzu. Detekce opakujících se vzorů v signálu.
  \end{Poznamka}
}

\QA{Co to znamená, když jsou dva signály vzájemně ortogonální?}{%
  Signály $s_1(t)$ a $s_2(t)$ jsou \textbf{vzájemně ortogonální}, pokud jejich vzájemná energie (resp.\ výkon) je nulová:
  \begin{Vzorec}[Podmínka ortogonality]
  \[
    E_{12} = \int_{-\infty}^{+\infty} s_1(t)\,s_2^*(t)\,\dd t = 0
    \quad \Leftrightarrow \quad s_1 \perp s_2
  \]
  \end{Vzorec}
  \textbf{Fyzikální význam}: ortogonální signály se \textit{vzájemně neovlivňují} – nemají žádný „energetický průnik".

  \textbf{Proč je ortogonalita důležitá?}
  \begin{itemize}[itemsep=3pt]
    \item Je \textbf{nutnou podmínkou jednoznačnosti} rozkladu signálu do bázových funkcí (Fourierova řada).
    \item Rozklad signálu je jednoznačný právě tehdy, když jsou bázové funkce ortogonální – nelze je zaměnit.
    \item Koeficienty FS jsou proto určeny jednoznačně: $c_n = \frac{1}{T_0}\int_{T_0} s(t)\,e^{-jn\omega_0 t}\,\dd t$.
  \end{itemize}
  \begin{Poznamka}[Příklady ortogonálních signálů]
    $\sin(n\omega_0 t) \perp \cos(m\omega_0 t)$ pro všechna $n, m$.\\
    $\sin(n\omega_0 t) \perp \sin(m\omega_0 t)$ pro $n \neq m$. \\
    Prakticky: OFDM subnosné, CDMA kódy, kvadraturní složky I/Q.
  \end{Poznamka}
}

\QA{Vysvětlete princip rozkladu periodického signálu do Fourierovy řady.}{%
  Každý periodický signál $s(t)$ s periodou $T_0$ lze vyjádřit jako \textbf{nekonečnou lineární kombinaci komplexních harmonických funkcí}:
  \begin{Vzorec}[Fourierova řada – syntéza a analýza]
  \[
    s(t) = \sum_{n=-\infty}^{+\infty} c_n\, e^{jn\omega_0 t},
    \qquad \omega_0 = \frac{2\pi}{T_0}
  \]
  \[
    c_n = \frac{1}{T_0}\int_{T_0} s(t)\, e^{-jn\omega_0 t}\,\dd t
  \]
  \end{Vzorec}
  \textbf{Princip krok za krokem}:
  \begin{enumerate}[itemsep=3pt]
    \item Bázové funkce $\{e^{jn\omega_0 t}\}_{n\in\mathbb{Z}}$ tvoří \textbf{úplný ortogonální systém} – vzájemně se neovlivňují, takže popis je jednoznačný.
    \item Koeficient $c_n$ udává, \textit{jak velký podíl $n$-té harmonické} (kmitočtu $n\omega_0$) je obsažen v signálu $s(t)$.
    \item Výpočet $c_n$: pronásobíme $s(t)$ komplexně sdruženou bázovou funkcí $e^{-jn\omega_0 t}$ a průměrujeme přes periodu. Dělení $1/T_0$ zajišťuje ortonormalitu.
    \item Výsledné spektrum je \textbf{diskrétní, neperiodické} (čárové spektrum na násobcích $\omega_0$).
  \end{enumerate}
  \begin{Vzorec}[Parsevalova věta – výkon přes koeficienty]
  \[
    P = \sum_{n=-\infty}^{+\infty}|c_n|^2 = R(0)
  \]
  \end{Vzorec}
  \begin{Poznamka}[Proč harmonické funkce?]
    Jsou periodické, snadno derivovatelné, ortogonální a tvoří úplný systém. Navíc jsou vlastními funkcemi LTI soustav – soustava mění jen amplitudu a fázi.
  \end{Poznamka}
}

\QA{Popište aplikaci Fourierovy transformace (FT) k výpočtu spektra obecných signálů ve spojitém čase.}{%
  \textbf{Fourierova transformace (FT)} zobecňuje principy FS i na \textbf{neperiodické spojité signály}. Je odvozena limitním přechodem: z periodického signálu vyřízneme část délky $T$, aproximujeme ji FS a provedeme $T \to \infty$ – základní kmitočet $\omega_0 \to 0$, spektrum se zahustí a přejde ze součtu na integrál:
  \begin{Vzorec}[Přímá Fourierova transformace]
  \[
    S(\omega) = \FT\{s(t)\} = \int_{-\infty}^{+\infty} s(t)\, e^{-j\omega t}\,\dd t
  \]
  \end{Vzorec}
  \begin{Vzorec}[Zpětná Fourierova transformace]
  \[
    s(t) = \FT^{-1}\{S(\omega)\} = \frac{1}{2\pi}\int_{-\infty}^{+\infty} S(\omega)\, e^{j\omega t}\,\dd\omega
  \]
  \end{Vzorec}
  \begin{itemize}[itemsep=3pt]
    \item Výsledné spektrum $S(\omega)$ je \textbf{spojité, neperiodické}.
    \item Pro reálný signál $s(t)\in\mathbb{R}$: amplitudové spektrum $|S(\omega)|$ je \textbf{sudé}, fázové spektrum $\arg S(\omega)$ je \textbf{liché}.
  \end{itemize}
  \begin{Vzorec}[Parsevalova věta pro FT]
  \[
    E = \int_{-\infty}^{+\infty}|s(t)|^2\,\dd t = \frac{1}{2\pi}\int_{-\infty}^{+\infty}|S(\omega)|^2\,\dd\omega
  \]
  \end{Vzorec}
  \begin{Poznamka}[FT vs.\ FS]
    FS: periodický signál $\to$ diskrétní spektrum. FT: obecný (neperiodický) signál $\to$ spojité spektrum. FT je zobecněním FS na celou třídu signálů.
  \end{Poznamka}
}

\QA{Jak vypadá spektrum Fourierovy transformace pro Diracův impuls, konstantní signál, vzorkovací funkci, obdélníkový impuls a harmonický signál?}{%
  \begin{center}
  \renewcommand{\arraystretch}{1.6}
  \begin{tabular}{ll p{5cm}}
  \toprule
  \textbf{Signál} $s(t)$ & \textbf{Spektrum} $S(\omega)$ & \textbf{Komentář} \\
  \midrule
  $\delta(t)$ & $1$ & Dirac má plochá spektrum – obsahuje všechny frekvence stejnou měrou \\
  $1$ (konstanta) & $2\pi\,\delta(\omega)$ & Konstantní signál má veškerý výkon na $\omega=0$ \\
  $\Sa(\omega_0 t)$ & $\dfrac{\pi}{\omega_0}\rect\!\!\left(\dfrac{\omega}{2\omega_0}\right)$ & Sinc v čase dává obdélník ve spektru \\
  $\rect(t/\tau)$ & $\tau\,\Sa(\omega\tau/2)$ & Obdélník v čase dává sinc ve spektru \\
  $\cos(\omega_0 t)$ & $\pi[\delta(\omega-\omega_0)+\delta(\omega+\omega_0)]$ & Dva Diraci na $\pm\omega_0$ \\
  $e^{j\omega_0 t}$ & $2\pi\,\delta(\omega-\omega_0)$ & Jeden Dirac na $\omega_0$ \\
  \bottomrule
  \end{tabular}
  \end{center}
  \begin{Poznamka}[Princip duality FT]
    FT je duální: obdélník $\leftrightarrow$ sinc, Dirac $\leftrightarrow$ konstanta. Pokud znáte FT jednoho páru, ihned víte i FT opačného.
  \end{Poznamka}
}

\QA{Jakou základní vlastnost má spektrum DtFT diskrétních signálů?}{%
  Diskrétní Fourierova transformace v čase (\textbf{DtFT}) přiřazuje diskrétnímu neperiodickému signálu spektrum:
  \begin{Vzorec}[DtFT]
  \[
    S(e^{j\Omega}) = \sum_{k=-\infty}^{\infty} s[k]\, e^{-j\Omega k}
    \qquad
    s[k] = \frac{1}{2\pi}\int_{-\pi}^{\pi} S(e^{j\Omega})\, e^{j\Omega k}\,\dd\Omega
  \]
  \end{Vzorec}
  \textbf{Základní vlastnost}: spektrum DtFT je \textbf{spojité a periodické} s periodou $2\pi$ (nebo $\omega_{sa}$ v rad/s):
  \[
    S(e^{j(\Omega + 2\pi)}) = S(e^{j\Omega})
  \]
  \textbf{Proč je spektrum periodické?} Diskrétní signál vznikl vzorkováním s kmitočtem $f_{sa}$ – ve spektrální oblasti se proto periodicky opakují kopie spektra s periodou $\omega_{sa}$. DtFT tuto periodicitu přímo zahrnuje do definice.
  \begin{Info}[Přehled – typ signálu vs.\ typ spektra]
    \begin{tabular}{ll}
      FT (spojitý neperiodický) & $\to$ spektrum: spojité, neperiodické \\
      DtFT (diskrétní neperiodický) & $\to$ spektrum: \textbf{spojité, periodické} \\
      FS (spojitý periodický) & $\to$ spektrum: diskrétní, neperiodické \\
      DFS (diskrétní periodický) & $\to$ spektrum: diskrétní, periodické \\
    \end{tabular}
  \end{Info}
}

\QA{Vysvětlete význam věty o Fourierově transformaci konvoluce dvou signálů.}{%
  \begin{Vzorec}[Konvoluční věta]
  \[
    \FT\{s_1(t) * s_2(t)\} = S_1(\omega)\cdot S_2(\omega)
  \]
  \[
    \FT\{s_1(t)\cdot s_2(t)\} = \frac{1}{2\pi}\,S_1(\omega) * S_2(\omega)
  \]
  \end{Vzorec}
  \textbf{Fyzikální/matematický smysl}: konvoluce v časové oblasti (operace náročná na výpočet) odpovídá pouhému \textit{násobení} ve spektrální oblasti (snadná operace), a naopak.

  \textbf{Klíčové důsledky a využití}:
  \begin{itemize}[itemsep=4pt]
    \item \textbf{LTI soustava}: výstup $y(t) = x(t)*h(t)$ odpovídá ve spektru $Y(\omega) = X(\omega)\cdot H(\omega)$. Filtraci stačí provést jako násobení spekter.
    \item \textbf{Vzorkování}: vzorkování = násobení Diracovým hřebenem v čase $\Rightarrow$ ve spektru konvoluce s hřebenem = periodizace spektra.
    \item \textbf{Modulace}: násobení nosnou v čase $\Rightarrow$ posun spektra do nosného kmitočtu.
    \item \textbf{FFT}: umožňuje rychlý výpočet konvoluce přes DFT: místo $O(N^2)$ operací jen $O(N\log N)$.
  \end{itemize}
}

\section{Vzorkování a interpolace}
% ============================================================

\QA{Vysvětlete ve spektrální oblasti odvození, význam a důsledky Shannonova vzorkovacího teorému.}{%
  \textbf{Formulace}: Spektrálně omezený signál s maximálním kmitočtem $f_m$ lze jednoznačně rekonstruovat ze svých vzorků, je-li splněna \textbf{Shannonova (Nyquistova) podmínka}:
  \begin{Vzorec}[Vzorkovací podmínka]
  \[
    f_{sa} > 2f_m \qquad \Leftrightarrow \qquad \omega_{sa} > 2\omega_m
  \]
  \end{Vzorec}
  \textbf{Odvození ve spektrální oblasti}:
  \begin{enumerate}[itemsep=4pt]
    \item \textbf{Ideální vzorkování} = násobení signálu Diracovým hřebenem:
      \[
        s_a(t) = s(t)\cdot\sum_{k=-\infty}^{\infty}\delta(t-kT_{sa})
      \]
    \item FT Diracova hřebene je opět hřeben s periodou $\omega_{sa}$. Proto:
      \[
        S_a(\omega) = \frac{1}{T_{sa}}\sum_{n=-\infty}^{\infty} S(\omega - n\omega_{sa})
      \]
      Spektrum vzorkovaného signálu je \textbf{periodická kopie} originálu s periodou $\omega_{sa}$.
    \item \textbf{Rekonstrukce}: je-li $\omega_{sa} > 2\omega_m$, kopie spektra se \textit{nepřekrývají}. Ideální dolní propustí s mezním kmitočtem $\omega_{sa}/2$ lze kopie oddělit a získat zpět $S(\omega)$.
    \item Je-li $\omega_{sa} \leq 2\omega_m$: kopie se překrývají – nastává \textbf{aliasing} (neodstranitelné zkreslení).
  \end{enumerate}
  \begin{Poznamka}[Aliasing – důsledek porušení podmínky]
    Při aliasingu se vysokofrekvenční složky signálu jeví jako nízkofrekvenční – dochází k nevratnému zkreslení. Prevence: anti-aliasing filtr (DP) před vzorkováním.
  \end{Poznamka}
}

\QA{Vysvětlete proces ideální interpolace vzorkovaného signálu.}{%
  \textbf{Ideální interpolace} = rekonstrukce spojitého signálu $s(t)$ ze vzorků $s[k] = s(kT_{sa})$.
  \begin{Vzorec}[Whittaker–Shannon – ideální interpolace]
  \[
    s(t) = \sum_{k=-\infty}^{\infty} s(kT_{sa})\cdot\Sa\!\left(\frac{\pi(t-kT_{sa})}{T_{sa}}\right)
  \]
  \end{Vzorec}
  \textbf{Postup}:
  \begin{enumerate}[itemsep=3pt]
    \item Vzorky $s[k]$ nahradíme váhovanými \textbf{Dirakovými impulsy} – vznikne ideálně vzorkovaný signál.
    \item Výsledek filtrujeme \textbf{ideální dolní propustí} s mezním kmitočtem $\omega_{sa}/2$ a zesílením $T_{sa}$.
    \item Impulsová odezva ideální DP je $\Sa$ funkce $\Rightarrow$ v čase dochází k interpolaci sincem.
  \end{enumerate}
  \textbf{Problémy realizace}: ideální DP je \textit{nekauzální} (nerealizovatelná) a pro generování Diracových impulsů nejsou fyzikální prostředky. V praxi se používají realizovatelné FIR/IIR filtry s přijatelným zkreslením.
}

\section{Klasifikace soustav, LTI, konvoluce, stabilita}
% ============================================================

\QA{Co je to soustava a operátor soustavy?}{%
  \textbf{Soustava} (systém) je objekt, který pod vlivem \textbf{buzení} (vstupního signálu $x$) produkuje \textbf{odezvu} (výstupní signál $y$):
  \[
    y = \mathcal{A}\{x\}
  \]
  $\mathcal{A}$ je \textbf{operátor soustavy} – matematický popis transformace vstupního signálu na výstupní.

  Soustava dává do vztahu skupinu signálů – skládá se z fyzikálních nebo matematických součástí. Může mít více vstupů a výstupů (MIMO).
  \begin{Info}[Příklady soustav]
    Filtry, zesilovače, přenosová vedení, modulátory, demodulátory, antény, přenosové kanály, enkodéry/dekodéry.
  \end{Info}
  \begin{Poznamka}[Smíšené soustavy]
    \textbf{Vzorkovač}: spojitý vstup $\to$ diskrétní výstup. \textbf{Interpolátor} (D/A): diskrétní vstup $\to$ spojitý výstup.
  \end{Poznamka}
}

\QA{Vysvětlete význam lineárních a časově invariantních soustav LTI.}{%
  Soustava $\mathcal{A}$ je \textbf{LTI} (Linear Time-Invariant), pokud splňuje dvě podmínky:

  \textbf{1) Linearita} – princip superpozice:
  \begin{Vzorec}[]
  \[
    \mathcal{A}\{\alpha x_1(t) + \beta x_2(t)\} = \alpha\,\mathcal{A}\{x_1(t)\} + \beta\,\mathcal{A}\{x_2(t)\}
  \]
  \end{Vzorec}
  Odezva na lineární kombinaci vstupů je lineární kombinací odezev na jednotlivé vstupy.

  \textbf{2) Časová invariance} – posun vstupu způsobí stejný posun výstupu:
  \begin{Vzorec}[]
  \[
    \mathcal{A}\{x(t-t_0)\} = y(t-t_0) \quad \forall\, t_0
  \]
  \end{Vzorec}
  \textbf{Proč jsou LTI soustavy tak důležité?}
  \begin{itemize}[itemsep=3pt]
    \item LTI soustava je \textbf{plně popsána} jedinou funkcí – impulsovou odezvou $h(t)$.
    \item Výstup je konvolucí: $y(t) = x(t) * h(t)$.
    \item Ve spektrální oblasti: $Y(\omega) = X(\omega)\cdot H(\omega)$ – velmi efektivní popis.
    \item Harmonická funkce je \textbf{vlastní funkcí} LTI soustavy: $e^{j\omega t} \to H(\omega)\,e^{j\omega t}$.
    \item Většina reálných soustav (filtry, obvody) je LTI nebo se jím blíží.
  \end{itemize}
}

\QA{Co je to impulsová odezva spojité a diskrétní LTI soustavy?}{%
  \textbf{Impulsová odezva} $h(t)$ (resp.\ $h[k]$) je odezva LTI soustavy na \textbf{Diracův impuls} (resp.\ jednotkový impuls) na vstupu:
  \begin{Vzorec}[]
  \[
    h(t) = \mathcal{A}\{\delta(t)\} \quad \text{(spojitý čas)}
    \qquad
    h[k] = \mathcal{A}\{\delta[k]\} \quad \text{(diskrétní čas)}
  \]
  \end{Vzorec}
  \textbf{Proč plně charakterizuje soustavu?} Protože díky linearitě a časové invarianci lze z $h(t)$ odvodit odezvu na \textit{libovolný} vstup $x(t)$ pomocí konvoluce:
  \[
    y(t) = x(t) * h(t) = \int_{-\infty}^{+\infty} x(\tau)\,h(t-\tau)\,\dd\tau
  \]
  \textbf{Kauzalita a stabilita čtená z $h(t)$}:
  \begin{itemize}[itemsep=3pt]
    \item \textbf{Kauzalita}: $h(t) = 0$ pro $t < 0$ – soustava nereaguje na budoucnost.
    \item \textbf{BIBO stabilita}: $\int_{-\infty}^{+\infty}|h(t)|\,\dd t < \infty$ (absolutní integrabilita).
  \end{itemize}
}

\QA{Vysvětlete princip a mechanismus výpočtu spojité a diskrétní konvoluce.}{%
  Konvoluce je operace, která dává výstup LTI soustavy pro daný vstup:
  \begin{Vzorec}[Spojitá konvoluce]
  \[
    y(t) = x(t) * h(t) = \int_{-\infty}^{+\infty} x(\tau)\,h(t-\tau)\,\dd\tau
  \]
  \end{Vzorec}
  \begin{Vzorec}[Diskrétní konvoluce]
  \[
    y[k] = x[k] * h[k] = \sum_{i=-\infty}^{\infty} x[i]\,h[k-i]
  \]
  \end{Vzorec}
  \textbf{Grafický postup} (pro každé $t$):
  \begin{enumerate}[itemsep=3pt]
    \item $x(\tau)$ – první signál jako funkce proměnné $\tau$.
    \item $h(-\tau)$ – druhý signál \textbf{převrátíme} (zrcadlení podle osy $\tau=0$).
    \item $h(t-\tau)$ – převrácený signál \textbf{posuneme doprava o $t$}.
    \item Integrál (plocha pod grafem) součinu $x(\tau)\cdot h(t-\tau)$ = hodnota $y(t)$.
    \item Opakujeme pro všechna $t$.
  \end{enumerate}
  Vlastnosti konvoluce: \textbf{komutativní} ($a*b = b*a$), asociativní, distributivní.\\
  Délka výstupu: pro konečné signály délek $N_1$ a $N_2$ je délka výstupu $N_1 + N_2 - 1$.
  \begin{Poznamka}[Výpočet přes FT]
    Konvoluci lze efektivně spočítat jako $y(t) = \FT^{-1}\{X(\omega)\cdot H(\omega)\}$ (rychlejší pro dlouhé signály díky FFT).
  \end{Poznamka}
}

\QA{Jak poznáme z impulsové odezvy soustavy, že jde o soustavu kauzální a stabilní?}{%
  Z impulsové odezvy $h(t)$ (resp.\ $h[k]$) lze přímo určit obě vlastnosti:
  \begin{Vzorec}[Kauzalita]
  \[
    h(t) = 0 \quad \text{pro } t < 0 \qquad \text{resp.} \qquad h[k] = 0 \quad \text{pro } k < 0
  \]
  \end{Vzorec}
  Soustava reaguje jen na \textit{současné nebo minulé} hodnoty vstupu – nezávisí na budoucnosti.

  \begin{Vzorec}[BIBO stabilita]
  \[
    \int_{-\infty}^{+\infty}|h(t)|\,\dd t < \infty
    \qquad \text{resp.} \qquad
    \sum_{k=-\infty}^{\infty}|h[k]| < \infty
  \]
  \end{Vzorec}
  Impulsová odezva musí být \textbf{absolutně integrabilní (sumovatelná)}.

  \textbf{Intuice}: je-li impulsová odezva příliš „velká" nebo trvá „příliš dlouho", soustava na omezený vstup může produkovat neomezený výstup – nestabilita.
  \begin{Info}[Marginální stabilita]
    Pokud $\int|h|\dd t = \infty$, ale soustava nevybuchuje pro ohraničený vstup – jde o \textit{marginálně stabilní} soustavu (napr.\ čistý integrátor). Taková soustava nesplňuje BIBO podmínku.
  \end{Info}
}

\QA{Co je to systémová funkce LTI soustavy a jaký je její vztah k impulsové odezvě?}{%
  \textbf{Systémová funkce} $H(s)$ (resp.\ $H(z)$) je \textbf{Laplaceova (resp.\ Z-) transformace impulsové odezvy}:
  \begin{Vzorec}[]
  \[
    H(s) = \LT\{h(t)\} = \int_{-\infty}^{+\infty} h(t)\,e^{-st}\,\dd t \quad \text{(spojitý čas)}
  \]
  \[
    H(z) = \ZT\{h[k]\} = \sum_{k=-\infty}^{\infty} h[k]\,z^{-k} \quad \text{(diskrétní čas)}
  \]
  \end{Vzorec}
  Pro LTI soustavu popsanou diferenciální (diferenční) rovnicí je $H(s)$ racionální lomená funkce:
  \[
    H(s) = \frac{b_M s^M + \cdots + b_0}{a_N s^N + \cdots + a_0} = \frac{B(s)}{A(s)}
  \]
  \textbf{Vztah k frekvenční charakteristice}:
  \[
    H(\omega) = H(s)\big|_{s=j\omega} = \FT\{h(t)\} \quad \text{(platí pro stabilní soustavy)}
  \]
  \begin{Poznamka}[Membránový model]
    Vizualizace $|H(s)|$ v komplexní rovině: nuly „přišpendlí" membránu k nule, póly ji „vytáhnou" nahoru. Přenosová funkce $|H(j\omega)|$ odpovídá výšce membrány podél imaginární osy.
  \end{Poznamka}
}

\QA{Co jsou nuly a póly systémové funkce a jak z pozice pólů poznáme, zda je soustava stabilní?}{%
  Pro systémovou funkci ve tvaru $H(s) = B(s)/A(s)$:
  \begin{itemize}[itemsep=4pt]
    \item \textbf{Nuly} – kořeny čitatele $B(s) = 0$: kmitočty, kde $|H(s)| = 0$ (soustava zcela potlačuje daný kmitočet).
    \item \textbf{Póly} – kořeny jmenovatele $A(s) = 0$: charakteristické frekvence soustavy, kde $|H(s)| \to \infty$.
  \end{itemize}
  \begin{Vzorec}[Stabilita z polohy pólů]
  \textbf{Spojitá soustava}: stabilní $\Leftrightarrow$ všechny póly v \textbf{levé polorovině}:
  \[
    \operatorname{Re}(p_i) < 0 \quad \forall\, i
  \]
  \textbf{Diskrétní soustava}: stabilní $\Leftrightarrow$ všechny póly \textbf{uvnitř jednotkové kružnice}:
  \[
    |p_i| < 1 \quad \forall\, i
  \]
  \end{Vzorec}
  \textbf{Proč?} Impulsová odezva spojité soustavy má tvar $h(t) \sim e^{p_i t}$. Je-li $\operatorname{Re}(p_i) < 0$, funkce exponenciálně klesá $\Rightarrow$ absolutně integrabilní $\Rightarrow$ stabilní.
  \begin{Info}[Marginální stabilita]
    Pól na imaginární ose ($\operatorname{Re}(p_i) = 0$) nebo na jednotkové kružnici: marginálně stabilní soustava (neodmocňuje, ale nevybuchne).
  \end{Info}
}

\QA{Jak určíme frekvenční charakteristiku (přenosovou funkci) LTI soustavy ze známé impulsové odezvy?}{%
  Frekvenční charakteristika $H(\omega)$ se získá jako \textbf{Fourierova transformace impulsové odezvy}:
  \begin{Vzorec}[Spojitá soustava]
  \[
    H(\omega) = \FT\{h(t)\} = \int_{-\infty}^{+\infty} h(t)\,e^{-j\omega t}\,\dd t = H(s)\big|_{s=j\omega}
  \]
  \end{Vzorec}
  \begin{Vzorec}[Diskrétní soustava]
  \[
    H(e^{j\Omega}) = \mathrm{DtFT}\{h[k]\} = \sum_{k=-\infty}^{\infty} h[k]\,e^{-j\Omega k} = H(z)\big|_{z=e^{j\Omega}}
  \]
  \end{Vzorec}
  Z frekvenční charakteristiky:
  \begin{itemize}[itemsep=3pt]
    \item $|H(\omega)|$ – \textbf{amplitudová charakteristika}: jak soustava mění amplitudu na kmitočtu $\omega$.
    \item $\arg H(\omega)$ – \textbf{fázová charakteristika}: fázový posun výstupu vůči vstupu.
    \item Pro reálnou $h(t)$: $|H(\omega)|$ je sudá, $\arg H(\omega)$ je lichá.
    \item Frekvenční charakteristika diskrétní soustavy je periodická s periodou $2\pi$ (resp.\ $\omega_{sa}$).
  \end{itemize}
  \textbf{Odezva na harmonické buzení} $x(t) = A\cos(\omega_0 t + \varphi)$:
  \[
    y(t) = A\cdot|H(\omega_0)|\cdot\cos\!\bigl(\omega_0 t + \varphi + \arg H(\omega_0)\bigr)
  \]
  LTI soustava mění jen \textit{amplitudu a fázi}, kmitočet zůstává stejný.
}

\section{Pásmové signály, komplexní obálka, Hilbertova transformace}
% ============================================================

\QA{Jaké specifické vlastnosti má reálný pásmový signál BP, zejména s ohledem na jeho spektrum?}{%
  \textbf{Pásmový signál} (Band-Pass, BP) má spektrum soustředěné \textbf{v okolí nosného kmitočtu $\omega_c$}:
  \begin{itemize}[itemsep=4pt]
    \item Spektrum je nenulové pouze v pásmech $[\omega_d, \omega_h]$ a $[-\omega_h, -\omega_d]$ (symetricky).
    \item \textbf{Symetrické amplitudové spektrum}: $|S(\omega)| = |S(-\omega)|$ – důsledek toho, že signál je reálný.
    \item Šířka pásma $B = \omega_h - \omega_d$; nosný kmitočet $\omega_c \approx (\omega_h+\omega_d)/2$.
    \item Typicky: $\omega_c \gg B$ (nosný kmitočet je mnohem vyšší než šířka pásma).
    \item Lze zapsat jako: $s(t) = V(t)\cos\!\bigl(\omega_c t + \Theta(t)\bigr)$, kde $V(t)$ je \textbf{obálka} a $\Theta(t)$ je \textbf{fáze}.
  \end{itemize}
  \begin{Poznamka}[Příklady BP signálů]
    AM rádio (kolem 1 MHz), FM rádio (kolem 100 MHz), mobilní signály (GHz pásmo). Všechny mají $\omega_c \gg B$.
  \end{Poznamka}
}

\QA{Popište ve spektrální oblasti přechod od reálného pásmového signálu přes analytický signál ke komplexní obálce CE.}{%
  \textbf{Motivace}: reálný BP signál má symetrické spektrum – záporné frekvence jsou redundantní. Stačí popsat signál jedním postranním pásmem – zavádíme komplexní obálku.

  \textbf{Krok 1 – Analytický signál} $z(t)$:
  \begin{Vzorec}[]
  \[
    Z(\omega) = 2\cdot\mathbf{1}(\omega)\cdot S(\omega)
    = \begin{cases} 2S(\omega) & \omega > 0 \\ S(0) & \omega = 0 \\ 0 & \omega < 0 \end{cases}
  \]
  \end{Vzorec}
  Oříznutí záporných frekvencí a zdvojení kladných. V čase: $z(t) = s(t) + j\hat{s}(t)$, kde $\hat{s}(t)$ je Hilbertova transformace $s(t)$. Analytický signál je \textit{komplexní}.

  \textbf{Krok 2 – Komplexní obálka} $\tilde{s}(t)$:
  \begin{Vzorec}[]
  \[
    \tilde{S}(\omega) = Z(\omega + \omega_c) \qquad \Leftrightarrow \qquad \tilde{s}(t) = z(t)\cdot e^{-j\omega_c t}
  \]
  \end{Vzorec}
  Posunutí spektra analytického signálu \textbf{dolů o $\omega_c$} do základního pásma. CE je \textbf{nízkofrekvenční komplexní reprezentace} VF pásmového signálu.

  \textbf{Zpětný přechod}:
  \[
    s(t) = \operatorname{Re}\!\bigl\{\tilde{s}(t)\cdot e^{j\omega_c t}\bigr\}
  \]
  Soufázová složka: $a(t) = \operatorname{Re}\{\tilde{s}\}$. \quad Kvadraturní složka: $c(t) = \operatorname{Im}\{\tilde{s}\}$.
  \begin{Info}[Výhody CE oproti BP signálu]
    Nižší vzorkovací kmitočet (signál je NF), jednodušší matematický popis, snadnější simulace a implementace algoritmů.
  \end{Info}
}

\QA{Co je Hilbertova transformace, jaké má základní vlastnosti a jaký je její význam?}{%
  \textbf{Hilbertova transformace} (HT) je LTI soustava, která všem frekvenčním složkám přidá fázový posun $-\pi/2$:
  \begin{Vzorec}[Přenosová funkce Hilbertova transformátoru]
  \[
    H(\omega) = -j\cdot\operatorname{sgn}(\omega)
    = \begin{cases} -j & \omega > 0 \\ 0 & \omega = 0 \\ +j & \omega < 0 \end{cases}
  \]
  \end{Vzorec}
  \begin{Vzorec}[Impulsová odezva]
  \[
    h(t) = \frac{1}{\pi t}
  \]
  \end{Vzorec}
  \textbf{Základní vlastnosti}:
  \begin{itemize}[itemsep=3pt]
    \item \textbf{Lineární} soustava (realizace konvolucí s $1/\pi t$).
    \item \textbf{Amplitudová charakteristika konstantní}: $|H(\omega)| = 1$ – HT \textit{nemění amplitudy}.
    \item Na kladných kmitočtech: \textbf{fázové zpoždění o $\pi/2$}.
    \item Na záporných kmitočtech: fázový předstih o $\pi/2$.
    \item Soustava je \textbf{nekauzální} ($h(t)\neq 0$ pro $t<0$) a \textbf{nestabilní} (není absolutně integrabilní).
    \item $\hat{\hat{s}}(t) = -s(t)$ – dvojitá HT vrátí invertovaný signál.
  \end{itemize}
  \textbf{Využití}: výpočet analytického signálu $z(t) = s(t) + j\hat{s}(t)$, kde $\hat{s}(t) = \mathcal{H}\{s(t)\}$. To je základ přechodu k CE. Také se používá v detekci obálky, SSB modulaci a zpracování obrazu.
}

\section{Analogové modulace}
% ============================================================

\QA{Vysvětlete základní význam modulací v komunikačním systému. Co je nosná vlna, modulační signál a modulovaný signál?}{%
  \textbf{Modulace} = ovlivnění parametrů nosné vlny modulačním signálem za účelem přenosu informace vhodnou formou:
  \[
    s(t) = V(t)\cdot\cos\!\bigl(\omega_c t + \Theta(t)\bigr)
  \]
  \begin{itemize}[itemsep=4pt]
    \item \textbf{Nosná vlna} $s_c(t) = A_c\cos(\omega_c t)$: vysokofrekvenční harmonický signál, který \textit{vyhovuje přenosovému médiu} (RF vlna, optický paprsek). Samotná nosná nese nulovou informaci.
    \item \textbf{Modulační signál} $s_M(t)$: nese informaci, kterou chceme přenést (hlas, data, video). Typicky nízkofrekvenční.
    \item \textbf{Modulovaný signál} $s(t)$: výstup modulátoru. Přenáší informaci v podobě vhodné pro dané médium (frekvenční pásmo, výkon).
  \end{itemize}
  \textbf{Proč modulujeme?}
  \begin{itemize}[itemsep=3pt]
    \item Přizpůsobení kmitočtového pásma přenosovému médiu (anténa musí mít rozměr $\sim\lambda/4$).
    \item Multiplexování více signálů do různých pásem (AM rádio, FM rádio, LTE).
    \item Odolnost vůči rušení a šumu.
  \end{itemize}
  \begin{Info}[Dělení modulací]
    \textbf{Amplitudové}: modulační signál ovlivňuje amplitudu $V(t)$, fáze $\Theta$ = konst.\\
    \textbf{Úhlové}: modulační signál ovlivňuje fázi/kmitočet $\Theta(t)$, obálka $V$ = konst.
  \end{Info}
}

\QA{Vysvětlete základní princip amplitudové modulace AM s nosnou vlnou v časové a spektrální oblasti.}{%
  \begin{Vzorec}[AM-DSB se zachovanou nosnou]
  \[
    s_{AM}(t) = A_c\bigl[1 + m\cdot s_M(t)\bigr]\cos(\omega_c t)
  \]
  kde $m$ je \textbf{modulační index} (hloubka modulace), $0 < m \leq 1$.
  \end{Vzorec}
  \textbf{Princip v časové oblasti}:
  \begin{itemize}[itemsep=3pt]
    \item Nosná vlna má vyšší kmitočet $\omega_c$.
    \item Amplituda nosné (obálka) se mění podle tvaru modulačního signálu $s_M(t)$.
    \item Modulovaný signál = nosná vlna s obálkou ve tvaru $s_M(t)$.
  \end{itemize}
  \textbf{Princip ve spektrální oblasti}:
  \begin{itemize}[itemsep=3pt]
    \item Spektrum $s_M(\omega)$ se posune do nosného kmitočtu $\omega_c$ (a $-\omega_c$) a zmenší na $mA_c/2$.
    \item Ve spektru jsou: \textbf{nosná složka} (Diraci na $\pm\omega_c$) + \textbf{horní postranní pásmo (USB)} + \textbf{dolní postranní pásmo (LSB)}.
    \item Obě postranní pásma nesou \textit{stejnou} informaci – plýtvání spektrem.
  \end{itemize}
  \textbf{Nevýhody AM-DSB}:
  \begin{itemize}[itemsep=2pt]
    \item Nízká energetická účinnost (většina výkonu v nosné vlně).
    \item Spektrální neefektivita (USB i LSB nesou totéž).
  \end{itemize}
  \textbf{Varianty}: AM-DSB-SC (potlačená nosná) $\to$ AM-SSB (jedno postranní pásmo) $\to$ AM-VSB (vestigiální, TV).
}

\QA{Vysvětlete základní princip úhlových modulací FM a PM. Jaký je vztah těchto modulací?}{%
  U \textbf{úhlových modulací} je obálka konstantní $V(t) = A_c$, mění se \textbf{fáze nebo okamžitý kmitočet}:

  \begin{Vzorec}[Fázová modulace PM (Phase Modulation)]
  \[
    s_{PM}(t) = A_c\cos\!\bigl(\omega_c t + k_P\cdot s_M(t)\bigr)
  \]
  $k_P$ – citlivost fázového modulátoru [rad / jednotka $s_M$].
  \end{Vzorec}

  \begin{Vzorec}[Kmitočtová modulace FM (Frequency Modulation)]
  \[
    s_{FM}(t) = A_c\cos\!\left(\omega_c t + k_F\int_{-\infty}^{t} s_M(\tau)\,\dd\tau\right)
  \]
  $k_F$ – citlivost kmitočtového modulátoru [rad/s / jednotka $s_M$].
  \end{Vzorec}

  \textbf{Vztah FM a PM}: okamžitý kmitočet je derivací fáze:
  \[
    \omega_i(t) = \frac{\dd\Phi(t)}{\dd t} = \omega_c + k_F\cdot s_M(t) \quad \text{(FM)}
    \qquad
    \omega_i(t) = \omega_c + k_P\cdot\dot{s}_M(t) \quad \text{(PM)}
  \]
  FM s \textit{integrovaným} modulačním signálem = PM. PM s \textit{derivovaným} vstupem = FM. Proto jsou \textbf{vzájemně zaměnitelné} s vhodným předzpracováním.

  \textbf{Výhody úhlových modulací}:
  \begin{itemize}[itemsep=3pt]
    \item Konstantní obálka – lze použít nelineární zesilovače třídy C (vysoká účinnost).
    \item Lepší odolnost vůči šumu oproti AM.
    \item Využití: FM rádio (88–108 MHz), přenos zvuku ve videu, GSM, navigace.
  \end{itemize}
}

\section{Náhodné procesy, stacionarita, ergodicita, bílý šum}
% ============================================================

\QA{Co je náhodný signál (náhodný proces) a jaký je jeho vztah k náhodnému jevu a náhodné veličině?}{%
  \textbf{Náhodný proces} $\xi(t, \zeta)$ je parametrizovaný soubor funkcí, kde $\zeta\in\Omega$ je náhodný jev z pravděpodobnostního prostoru $(\Omega, \mathcal{F}, P)$:
  \begin{itemize}[itemsep=4pt]
    \item Pro fixní $\zeta = \zeta_i$: $\xi(t, \zeta_i)$ je jedna \textbf{realizace} – deterministický signál (možný průběh signálu).
    \item Pro fixní $t = t_0$: $\xi(t_0, \zeta)$ je \textbf{náhodná veličina} – hodnota procesu v čase $t_0$ je náhodná.
    \item Náhodné signály mají náhodný (nikoli chaotický) průběh – popisujeme je \textit{pravděpodobnostně}.
    \item Obsahují \textbf{neznámou informaci} nebo jsou výsledkem neznámých vlivů (šum, tepelný pohyb, atmosférická turbulence).
  \end{itemize}
  \textbf{Vztah k pojmům}:
  \begin{itemize}[itemsep=3pt]
    \item \textbf{Náhodný jev} $\zeta$: parametrizuje realizaci (např.\ „$i$-tý experiment").
    \item \textbf{Náhodná veličina}: hodnota procesu v daném čase.
    \item \textbf{Realizace}: konkrétní průběh pro dané $\zeta_i$ – deterministická funkce.
  \end{itemize}
  \begin{Poznamka}[Příklady]
    Tepelný šum v rezistoru, atmosférický šum, mluvený hlas, burzovní data, výsledky měření.
  \end{Poznamka}
}

\QA{Čím se vyznačují stacionární náhodné signály?}{%
  \textbf{Stacionární} náhodný proces = jeho \textbf{pravděpodobnostní charakteristiky nezávisí na volbě počátku časové osy} – statistické vlastnosti jsou časově neměnné.

  \begin{Info}[SSS – Stacionarita v užším smyslu (Strict Sense Stationary)]
    \textbf{Kompletní} statistický popis (hustota $n$-tého řádu) je invariantní vůči libovolnému časovému posunu:
    \[
      f_\xi(x_1,\ldots,x_n;\, t_1,\ldots,t_n) = f_\xi(x_1,\ldots,x_n;\, t_1+\Delta,\ldots,t_n+\Delta)
    \]
  \end{Info}

  \begin{Info}[WSS – Stacionarita v širším smyslu (Wide Sense Stationary)]
    Jen \textbf{vybrané charakteristiky} jsou invariantní:
    \[
      \mu = E[\xi(t)] = \text{konst.}
      \qquad
      R_\xi(t_1, t_2) = R_\xi(\tau), \quad \tau = t_2 - t_1
    \]
    Střední hodnota je konstantní a ACF závisí jen na rozdílu časů $\tau$.
  \end{Info}

  Platí: SSS $\Rightarrow$ WSS, ale WSS $\not\Rightarrow$ SSS.

  \textbf{Praktický dopad}: pro WSS procesy platí \textbf{Wiener-Chinčinova věta} – spektrální hustota výkonu $S_\xi(\omega)$ je FT autokorelační funkce:
  \[
    S_\xi(\omega) = \FT\{R_\xi(\tau)\}
  \]
}

\QA{Jaké vlastnosti má bílý šum, zejména s ohledem na spektrální hustotu výkonu a autokorelační funkci?}{%
  \textbf{Bílý šum} (White Noise, WN) je idealizovaný model náhodného signálu s \textbf{konstantní spektrální hustotou výkonu}:
  \begin{Vzorec}[Bílý šum]
  \[
    S_\xi(\omega) = S_0 = \text{konst.} \quad \forall\,\omega
  \]
  \[
    R_\xi(\tau) = \FT^{-1}\{S_0\} = S_0\,\delta(\tau)
  \]
  \end{Vzorec}
  \textbf{Vlastnosti}:
  \begin{itemize}[itemsep=4pt]
    \item \textbf{Konstantní spektrum}: všechny frekvence zastoupeny stejnou měrou (analogie bílého světla).
    \item \textbf{ACF = Diracův impuls v $\tau=0$}: hodnoty procesu v různých časech jsou \textit{nekorelované} – žádná „paměť".
    \item \textbf{Nekonečný výkon}: $P = \int_{-\infty}^{\infty} S_0\,\dd\omega = \infty$ – fyzikálně nerealizovatelný, jde o matematický ideál.
    \item \textbf{Nulová střední hodnota}: $E[\xi(t)] = 0$.
  \end{itemize}
  \begin{Info}[AWGN – Additive White Gaussian Noise]
    Model šumu v komunikačních systémech:
    \begin{itemize}[itemsep=2pt]
      \item \textbf{Aditivní}: $r(t) = s(t) + n(t)$ – šum se přičítá k signálu.
      \item \textbf{Bílý}: $S_n(\omega) = N_0/2$ (konstantní jednostranná hustota výkonu).
      \item \textbf{Gaussovský}: PDF hodnot je normální rozdělení:
        $f_n(x) = \frac{1}{\sqrt{2\pi\sigma^2}}\exp\!\left(-\frac{x^2}{2\sigma^2}\right)$
      \item Nekorelovaný – vzorky v různých časech jsou statisticky \textit{nezávislé}.
    \end{itemize}
  \end{Info}
}

\QA{Co jsou ergodické náhodné signály?}{%
  \textbf{Ergodické signály} – třída WSS náhodných signálů, pro které je \textbf{časové průměrování po jedné realizaci} ekvivalentní \textbf{průměrování přes množinu realizací} (ensemble averaging) v daném čase:
  \begin{Vzorec}[Ergodická střední hodnota]
  \[
    \mu = E[\xi(t)] = \lim_{T\to\infty}\frac{1}{2T}\int_{-T}^{T}\xi(t,\zeta_i)\,\dd t
    \quad \text{pro (skoro) všechna } \zeta_i
  \]
  \end{Vzorec}
  \begin{Vzorec}[Ergodická ACF]
  \[
    R_\xi(\tau) = \lim_{T\to\infty}\frac{1}{2T}\int_{-T}^{T}\xi(t+\tau,\zeta_i)\,\xi(t,\zeta_i)\,\dd t
  \]
  \end{Vzorec}
  \textbf{Klíčové vlastnosti}:
  \begin{itemize}[itemsep=4pt]
    \item Časová charakteristika jedné realizace je \textbf{determinovaná} (nulový rozptyl mezi realizacemi).
    \item \textbf{Nepotřebujeme průměrovat přes množinu realizací} – stačí jedna \textit{dostatečně dlouhá} realizace.
    \item Praktický dopad: statistiky signálu lze změřit z \textit{jediného záznamu} (jediného experimentu) $\Rightarrow$ zásadní pro praxi.
  \end{itemize}
  \begin{Poznamka}[Příklad neergo\-dického procesu]
    Náhodný proces, kde každá realizace má jinou konstantní amplitudu (náhodnou veličinou je jen amplituda). Průměr v čase $\neq$ průměr přes realizace.
  \end{Poznamka}
}
